CREATE TABLE supplier (
    id_supplier SERIAL PRIMARY KEY,
    nama VARCHAR(50) NOT NULL,
    alamat VARCHAR(50) NOT NULL,
    no_hp VARCHAR(13) NOT NULL UNIQUE,
    email VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE kategori_produk (
    id_kategori_produk SERIAL PRIMARY KEY,
    nama_kategori VARCHAR(30) NOT NULL
);

CREATE TABLE produk (
    id_produk Serial PRIMARY KEY,
    harga int NOT NULL,
    nama_produk VARCHAR(50) NOT NULL,
    stok int not null,
    status boolean not null,
    gambar text not null,
    deskripsi text,
    id_kategori_produk int not null REFERENCES kategori_produk(id_kategori_produk)
);

CREATE TABLE supplier_produk (
    id_supplier_produk serial PRIMARY KEY,
    id_supplier int NOT NULL REFERENCES supplier(id_supplier),
    id_produk int NOT NULL REFERENCES produk(id_produk)
);

CREATE TABLE log_stok (
    id_stok serial PRIMARY KEY,
    jumlah_stok int NOT NULL,
    kode_aksi int NOT NULL,
    updated_at DATE not null,
    id_produk int NOT NULL REFERENCES produk(id_produk)
);

CREATE TABLE pengiriman (
    id_pengiriman serial PRIMARY KEY,
    tanggal_kirim DATE NOT NULL,
    status_pengiriman boolean,
    id_user int NOT NULL REFERENCES users(id_user)
);

CREATE TABLE orders (
    id_order serial PRIMARY KEY,
    tanggal_order DATE NOT NULL,
    status boolean not null,
    metode_pembayaran boolean not null,
    tanggal_bayar DATE not null,
    id_customer int NOT NULL REFERENCES customer(id_customer),
    id_pengiriman int NOT NULL REFERENCES pengiriman(id_pengiriman),
    id_user int NOT NULL REFERENCES users(id_user)
);

CREATE TABLE detail_orders (
    id_detail_orders serial PRIMARY KEY,
    harga int NOT NULL,
    jumlah int NOT NULL,
    id_produk int NOT NULL REFERENCES produk(id_produk),
    id_order int NOT NULL REFERENCES orders(id_order)
);

CREATE TABLE kewenangan (
    id_kewenangan serial PRIMARY KEY,
    id_role int NOT NULL REFERENCES roles(id_role),
    id_user int NOT NULL REFERENCES users(id_user)
);

CREATE TABLE log_perubahan_user (
    id_log_perubahan_user serial PRIMARY KEY,
    kode_aksi int NOT NULL,
    data_lama text,
    data_baru text,
    update_at DATE not null,
    id_user int NOT NULL REFERENCES users(id_user)
);

CREATE TABLE customer (
    id_customer serial PRIMARY KEY,
    nama_customer VARCHAR(50) NOT NULL,
    no_hp VARCHAR(13) NOT NULL,
    alamat VARCHAR(50) not null
);

CREATE TABLE users (
    id_user serial PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    password VARCHAR(50) NOT NULL,
    nama VARCHAR(50) NOT NULL,
    no_hp VARCHAR(13) UNIQUE not null,
    alamat VARCHAR(50) not null,
    email VARCHAR(50) UNIQUE not null,
    is_active boolean not null,
	is_ready boolean not null
);

CREATE TABLE roles (
    id_role serial PRIMARY KEY,
    nama_role VARCHAR(30) NOT NULL
);

CREATE OR REPLACE FUNCTION fn_buat_order_baru(
    p_id_user INT,
    p_metode_pembayaran VARCHAR(50),
    p_metode_pengiriman VARCHAR(50)
)
RETURNS INT AS $$
DECLARE
    new_order_id INT;
BEGIN
    -- Masukkan data nota induk
    INSERT INTO orders (tanggal_order, status, metode_pembayaran, metode_pengiriman, tanggal_bayar, id_user)
    VALUES (CURRENT_DATE, 'Selesai', p_metode_pembayaran, p_metode_pengiriman, CURRENT_DATE, p_id_user)
    RETURNING id_order INTO new_order_id;

    -- Kembalikan ID order yang baru dibuat ke C#
    RETURN new_order_id;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_tambah_detail_order(
    p_id_order INT,
    p_id_produk INT,
    p_jumlah INT,
    p_harga NUMERIC
)
RETURNS VOID AS $$
BEGIN	
    -- 1. Masukkan barang ke tabel detail
    INSERT INTO detail_orders (id_order, id_produk, jumlah, harga)
    VALUES (p_id_order, p_id_produk, p_jumlah, p_harga);

    -- 2. Langsung kurangi stok produk (Ini alasan kenapa stokmu kemarin tidak berkurang!)
    UPDATE produk
    SET stok = stok - p_jumlah 
    WHERE id_produk = p_id_produk;
END;
$$ LANGUAGE plpgsql;

-- View riwayat penjualan untuk dashboard admin/kasir
CREATE OR REPLACE VIEW v_riwayat_penjualan AS
SELECT
    o.id_order,
    o.tanggal_order AS "Tanggal Order",
    u.nama AS "Kasir",
    c.nama_customer AS "Customer",
    p.nama_produk AS "Produk",
    d.jumlah AS "Qty",
    d.harga AS "Harga",
    (d.jumlah * d.harga) AS "Total",
    o.metode_pembayaran AS "Metode Bayar"
FROM orders o
JOIN detail_orders d ON o.id_order = d.id_order
JOIN produk p ON d.id_produk = p.id_produk
JOIN users u ON o.id_user = u.id_user
LEFT JOIN customer c ON o.id_customer = c.id_customer;

-- Function tambah user baru (karyawan)
CREATE OR REPLACE FUNCTION fn_tambah_user(
    p_username VARCHAR,
    p_password VARCHAR,
    p_nama VARCHAR,
    p_no_hp VARCHAR,
    p_alamat VARCHAR,
    p_email VARCHAR
)
RETURNS INT AS $$
DECLARE
    new_user_id INT;
BEGIN
    INSERT INTO users (username, password, nama, no_hp, alamat, email, is_active, is_ready)
    VALUES (p_username, p_password, p_nama, p_no_hp, p_alamat, p_email, true, false)
    RETURNING id_user INTO new_user_id;

    RETURN new_user_id;
END;
$$ LANGUAGE plpgsql;


-- Function ambil info user berdasarkan id
CREATE OR REPLACE FUNCTION fn_get_user_info(p_id_user INT)
RETURNS TABLE(
    id_user INT,
    nama VARCHAR,
    no_hp VARCHAR,
    alamat VARCHAR,
    email VARCHAR
) AS $$
BEGIN
    RETURN QUERY
    SELECT u.id_user, u.nama, u.no_hp, u.alamat, u.email
    FROM users u
    WHERE u.id_user = p_id_user;
END;
$$ LANGUAGE plpgsql;

CREATE TABLE pengiriman (
    id_pengiriman SERIAL PRIMARY KEY,
    alamat TEXT NOT NULL,                     -- Tambahan untuk menyimpan alamat tujuan
    status_pengiriman VARCHAR(50) DEFAULT 'Proses', -- Sesuai ERD: status_pengiriman
    tanggal_kirim DATE DEFAULT CURRENT_DATE,  -- Sesuai ERD: tanggal_kirim
    id_order INTEGER NOT NULL,                -- Foreign Key ke tabel order
    id_user INTEGER NOT NULL,                 -- Foreign Key ke tabel user (Kurir yang dikasih tugas)
    
    -- Menghubungkan Foreign Key ke tabel masing-masing
    CONSTRAINT fk_pengiriman_order FOREIGN KEY (id_order) REFERENCES "orders" (id_order) ON DELETE CASCADE,
    CONSTRAINT fk_pengiriman_user FOREIGN KEY (id_user) REFERENCES "users" (id_user)
);
