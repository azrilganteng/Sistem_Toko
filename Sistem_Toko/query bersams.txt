-- =====================================================
-- PROCEDURE: Transaksi Penjualan (ACID Transaction)
-- =====================================================
-- Atomicity:   Semua operasi (order + pengiriman) terjadi sebagai satu kesatuan.
--              Jika salah satu gagal, semua di-rollback.
-- Consistency: Memastikan id_customer, id_user, dan id_pengiriman valid
--              sesuai constraint database.
-- Isolation:   Berjalan dalam transaction context dari pemanggil,
--              tidak terganggu transaksi lain.
-- Durability:  Setelah COMMIT dari pemanggil, data permanen.
-- =====================================================

-- Procedure: Transaksi Penjualan Lengkap (ACID)
-- Atomicity:   Order + pengiriman + detail + stok dalam 1 procedure.
-- Consistency: Constraint database menjaga validitas data.
-- Isolation:   Berjalan dalam transaction context dari pemanggil.
-- Durability:  Setelah COMMIT dari pemanggil, data permanen.
--
-- Catatan: id_pengiriman di tabel orders harus nullable:
-- ALTER TABLE orders ALTER COLUMN id_pengiriman DROP NOT NULL;
--
CREATE OR REPLACE PROCEDURE p_transaksi_penjualan(
    p_id_user           INT,            -- Kasir ID
    p_id_customer       INT,            -- Customer ID
    p_metode_bayar      BOOLEAN,        -- true = Cash, false = Transfer
    OUT p_id_order      INT,
    p_id_kurir          INT DEFAULT NULL, -- Kurir ID (NULL = Langsung, ada value = Dikirim)
    p_produk_ids        INT[] DEFAULT NULL,  -- Array: id_produk
    p_produk_jumlahs    INT[] DEFAULT NULL,  -- Array: jumlah
    p_produk_hargas     NUMERIC[] DEFAULT NULL -- Array: harga
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_id_pengiriman INT := NULL;
    v_subtotal      NUMERIC;
    v_stok          INT;
    v_is_kurir      BOOLEAN;
BEGIN
    -- ============================================================
    -- STEP 1: Buat pengiriman (hanya jika ada kurir / Dikirim)
    -- Langsung = tidak insert pengiriman, id_pengiriman = NULL
    -- Validasi: kurir harus ada di users dengan role Kurir
    -- ============================================================
    IF p_id_kurir IS NOT NULL THEN
        -- Cek apakah user ini adalah kurir (validasi dari tabel users + kewenangan)
        SELECT EXISTS(
            SELECT 1 FROM users u
            JOIN kewenangan k USING(id_user)
            WHERE u.id_user = p_id_kurir
              AND k.id_role = 3
              AND u.is_active = true
              AND u.is_ready = true
        ) INTO v_is_kurir;

        IF NOT v_is_kurir THEN
            RAISE EXCEPTION 'User ID % bukan kurir yang valid atau tidak ready', p_id_kurir;
        END IF;

        INSERT INTO pengiriman (tanggal_kirim, status_pengiriman, id_user)
        VALUES (CURRENT_DATE, false, p_id_kurir)
        RETURNING id_pengiriman INTO v_id_pengiriman;
    END IF;

    -- ============================================================
    -- STEP 2: Buat order
    -- Tidak ada kolom metode_kirim di database.
    -- Langsung: id_pengiriman = NULL
    -- Dikirim:  id_pengiriman = v_id_pengiriman
    -- ============================================================
    INSERT INTO orders (tanggal_order, status, metode_pembayaran,
                        tanggal_bayar, id_user, id_customer, id_pengiriman)
    VALUES (CURRENT_DATE, false, p_metode_bayar,
            CURRENT_DATE, p_id_user, p_id_customer, v_id_pengiriman)
    RETURNING id_order INTO p_id_order;

    -- ============================================================
    -- STEP 3: Tambah detail order + kurangi stok
    -- Cek stok dulu, RAISE EXCEPTION jika tidak cukup
    -- ============================================================
    IF p_produk_ids IS NOT NULL THEN
        FOR i IN 1..array_length(p_produk_ids, 1) LOOP
            v_subtotal := p_produk_jumlahs[i] * p_produk_hargas[i];

            -- Cek stok (Consistency: stok tidak boleh negatif)
            SELECT stok INTO v_stok FROM produk WHERE id_produk = p_produk_ids[i];
            IF v_stok < p_produk_jumlahs[i] THEN
                RAISE EXCEPTION 'Stok tidak cukup untuk produk ID %. Tersedia: %, Diminta: %',
                    p_produk_ids[i], v_stok, p_produk_jumlahs[i];
            END IF;

            -- Insert detail
            INSERT INTO detail_orders (id_order, id_produk, jumlah, harga)
            VALUES (p_id_order, p_produk_ids[i], p_produk_jumlahs[i], v_subtotal);

            -- Kurangi stok
            UPDATE produk
            SET stok = stok - p_produk_jumlahs[i]
            WHERE id_produk = p_produk_ids[i];
        END LOOP;
    END IF;
END;
$$;

-- ============================================================
-- LOG STOK: Trigger untuk tracking perubahan stok
-- Tabel log_stok sudah ada (line 108-114)
-- kode_aksi: 1 = MASUK (restock), 2 = KELUAR (penjualan)
-- ============================================================

-- Trigger function: otomatis jalan setiap kali stok berubah
-- 1. Hapus trigger dan function lama agar bersih
DROP TRIGGER IF EXISTS trg_log_perubahan_user ON users;
DROP FUNCTION IF EXISTS fn_log_perubahan_user();

-- 2. Buat ulang fungsi triggernya
CREATE OR REPLACE FUNCTION fn_log_perubahan_user()
RETURNS TRIGGER AS $$
BEGIN
    -- 1 = INSERT
    IF TG_OP = 'INSERT' THEN
        INSERT INTO log_perubahan_user (kode_aksi, data_lama, data_baru, update_at)
        VALUES (
            1, 
            NULL,
            ROW(NEW.id_user, NEW.username, NEW.nama, NEW.no_hp, NEW.alamat, NEW.email, NEW.is_active)::TEXT,
            CURRENT_DATE
        );

    -- 2 = UPDATE, 3 = SOFT DELETE
    ELSIF TG_OP = 'UPDATE' THEN
        -- cek apakah semua data persis sama, jika iya jangan catat
        IF OLD.username = NEW.username AND
           OLD.password = NEW.password AND
           OLD.nama     = NEW.nama     AND
           OLD.no_hp    = NEW.no_hp    AND
           OLD.alamat   = NEW.alamat   AND
           OLD.email    = NEW.email    AND
           OLD.is_active = NEW.is_active THEN
            RETURN NEW; 
        END IF;

        -- soft delete (is_active berubah dari true ke false)
        IF OLD.is_active = true AND NEW.is_active = false THEN
            INSERT INTO log_perubahan_user (kode_aksi, data_lama, data_baru, update_at)
            VALUES (
                3,
                ROW(OLD.id_user, OLD.username, OLD.nama, OLD.no_hp, OLD.alamat, OLD.email, OLD.is_active)::TEXT,
                NULL,
                CURRENT_DATE
            );
        ELSE
            -- update biasa
            INSERT INTO log_perubahan_user (kode_aksi, data_lama, data_baru, update_at)
            VALUES (
                2,
                ROW(OLD.id_user, OLD.username, OLD.nama, OLD.no_hp, OLD.alamat, OLD.email, OLD.is_active)::TEXT,
                ROW(NEW.id_user, NEW.username, NEW.nama, NEW.no_hp, NEW.alamat, NEW.email, NEW.is_active)::TEXT,
                CURRENT_DATE
            );
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 3. Pasang/Attach trigger-nya ke tabel users
CREATE TRIGGER trg_log_perubahan_user
    AFTER INSERT OR UPDATE ON users
    FOR EACH ROW
    EXECUTE FUNCTION fn_log_perubahan_user();


CREATE OR REPLACE FUNCTION fn_log_perubahan_user()
RETURNS TRIGGER AS $$
BEGIN
    -- 1 = INSERT
    IF TG_OP = 'INSERT' THEN
        -- Perhatikan tambahan id_user dan NEW.id_user di bawah ini
        INSERT INTO log_perubahan_user (id_user, kode_aksi, data_lama, data_baru, update_at)
        VALUES (
            NEW.id_user,
            1, 
            NULL,
            ROW(NEW.id_user, NEW.username, NEW.nama, NEW.no_hp, NEW.alamat, NEW.email, NEW.is_active)::TEXT,
            CURRENT_DATE
        );

    -- 2 = UPDATE, 3 = SOFT DELETE
    ELSIF TG_OP = 'UPDATE' THEN
        -- cek apakah semua data persis sama, jika iya jangan catat
        IF OLD.username = NEW.username AND
           OLD.password = NEW.password AND
           OLD.nama     = NEW.nama     AND
           OLD.no_hp    = NEW.no_hp    AND
           OLD.alamat   = NEW.alamat   AND
           OLD.email    = NEW.email    AND
           OLD.is_active = NEW.is_active THEN
            RETURN NEW; 
        END IF;

        -- soft delete (is_active berubah dari true ke false)
        IF OLD.is_active = true AND NEW.is_active = false THEN
            INSERT INTO log_perubahan_user (id_user, kode_aksi, data_lama, data_baru, update_at)
            VALUES (
                OLD.id_user,
                3,
                ROW(OLD.id_user, OLD.username, OLD.nama, OLD.no_hp, OLD.alamat, OLD.email, OLD.is_active)::TEXT,
                NULL,
                CURRENT_DATE
            );
        ELSE
            -- update biasa
            INSERT INTO log_perubahan_user (id_user, kode_aksi, data_lama, data_baru, update_at)
            VALUES (
                NEW.id_user,
                2,
                ROW(OLD.id_user, OLD.username, OLD.nama, OLD.no_hp, OLD.alamat, OLD.email, OLD.is_active)::TEXT,
                ROW(NEW.id_user, NEW.username, NEW.nama, NEW.no_hp, NEW.alamat, NEW.email, NEW.is_active)::TEXT,
                CURRENT_DATE
            );
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 3. Pasang/Attach trigger-nya ke tabel users
CREATE OR REPLACE TRIGGER trg_log_perubahan_user
    AFTER INSERT OR UPDATE ON users
    FOR EACH ROW
    EXECUTE FUNCTION fn_log_perubahan_user();


CREATE OR REPLACE VIEW v_detail_order AS
SELECT
    d.id_order,
    p.nama_produk,
    d.jumlah,
    d.harga,
    (d.jumlah * d.harga) AS subtotal
FROM detail_orders d
JOIN produk p
    ON d.id_produk = p.id_produk;

CREATE OR REPLACE VIEW v_customer AS
SELECT id_customer, nama_customer, no_hp, alamat 
FROM customer 
ORDER BY nama_customer

CREATE OR REPLACE FUNCTION fn_tambah_produk(
    p_nama_produk       VARCHAR,
    p_harga             INT,
    p_stok              INT,
    p_deskripsi         TEXT,
    p_gambar            VARCHAR,
    p_id_kategori_produk INT
)
RETURNS INT
LANGUAGE plpgsql
AS $$
DECLARE
    v_id_produk INT;
BEGIN
    INSERT INTO produk (nama_produk, harga, stok, deskripsi, gambar, id_kategori_produk, status)
    VALUES (p_nama_produk, p_harga, p_stok, p_deskripsi, p_gambar, p_id_kategori_produk, true)
    RETURNING id_produk INTO v_id_produk;

    -- Trigger trg_log_stok akan otomatis mencatat ke log_stok
    RETURN v_id_produk;
END;
$$;

CREATE OR REPLACE VIEW v_profil_user AS
SELECT u.id_user, u.username, u.password, u.nama, u.email, u.no_hp, u.alamat, u.is_active, u.is_ready,
       STRING_AGG(r.nama_role, ', ') AS roles
FROM users u
LEFT JOIN kewenangan k ON k.id_user = u.id_user
LEFT JOIN roles r ON r.id_role = k.id_role
GROUP BY u.id_user, u.username, u.password, u.nama, u.email, u.no_hp, u.alamat, u.is_active, u.is_ready;

)
